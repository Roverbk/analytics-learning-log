### Final query (BigQuery, transactions CTE = setup data)
```sql
-- Input grain: one row = one transaction (a card can have 2+ txns on the same day)
-- Output grain: one row = one card (card_id) with its longest streak
-- Window needed? yes: ROW_NUMBER to label islands, ROW_NUMBER to pick the longest
-- Keep a streak when: it is the card's longest; if tied, the earliest

WITH transactions AS ( /* setup data */ ),
card_days AS (
  SELECT DISTINCT card_id, txn_date FROM transactions
),
numbered AS (   -- no tie-breaker needed: (card_id, txn_date) unique after dedupe
  SELECT card_id, txn_date,
         ROW_NUMBER() OVER (PARTITION BY card_id ORDER BY txn_date) AS day_num
  FROM card_days
),
islands AS (
  SELECT card_id, txn_date,
         DATE_SUB(txn_date, INTERVAL day_num DAY) AS island_label
  FROM numbered
),
streaks AS (
  SELECT card_id, island_label,
         COUNT(*) AS streak_length,
         MIN(txn_date) AS streak_start,
         MAX(txn_date) AS streak_end
  FROM islands
  GROUP BY card_id, island_label
),
ranked_streaks AS (   -- tie-breaker: earliest start
  SELECT *,
         ROW_NUMBER() OVER (PARTITION BY card_id
                            ORDER BY streak_length DESC, streak_start) AS streak_rank
  FROM streaks
)
SELECT card_id, streak_length, streak_start, streak_end
FROM ranked_streaks
WHERE streak_rank = 1
ORDER BY card_id;
```
